//===== transcendent-third-classes: Royal Guard ===========
//= The Crusader's other transcendent path.
//=
//= After the Valkyrie's rebirth, a High Swordman of job 40 or more
//= whose rebirth target is Paladin (ADVJOB) may become a
//= Royal Guard here instead, beside the stock Paladin changer,
//= which is untouched. The choice is permanent.
//=
//= Royal Guard is a third class in rAthena. The change goes
//= through Paladin and at once on to Royal Guard, so
//= rAthena's third-class skill lock records Paladin job 1
//= and holds nothing back. The class's tree is renewal's
//= non-transcendent Royal Guard tree (Novice, Swordman,
//= Crusader and its own skills), with the same 69 job levels a
//= Paladin has: a sidegrade, not an upgrade.
//==============================================================

valkyrie,55,39,3	script	Royal Guard#Valkyrie	752,{
	.@n$ = "[Royal Guard]";
	if (Class == Job_Royal_Guard_T) {
		mes .@n$;
		mes "You chose the shield. Spears and shields made for the Guard were lost with the knights of old; look for them among the dead who still keep their posts. The Peco Peco Breeder will give you your gryphon.";
		close;
	}
	if (ADVJOB != Job_Paladin || Class != Job_Swordman_High) {
		mes .@n$;
		mes "A Crusader who has been reborn chooses here: the Paladin's path, or mine.";
		close;
	}
	if (JobLevel < 40) {
		mes .@n$;
		mes "Come back at job level 40 or later.";
		close;
	}
	if (SkillPoint) {
		mes .@n$;
		mes "Spend your skill points first.";
		close;
	}
	mes .@n$;
	mes "A Paladin strikes and sacrifices: Rapid Smiting, Martyr's Reckoning, Battle Chant. A ^0000FFRoyal Guard^000000 leaves them behind to hold the line: sweeping spears, shield blows, and a shield over the whole party.";
	next;
	mes .@n$;
	mes "You keep every Crusader skill. You will never learn a Paladin skill: no Rapid Smiting, no Martyr's Reckoning, no Gloria Domini, no Battle Chant.";
	mes "Your job levels, your body and your gear are the same as a Paladin's.";
	next;
	mes .@n$;
	mes "^FF0000This choice is permanent.^000000";
	if (select("I take the shield.:Not yet.") == 2)
		close;
	if (select("Permanently. I am sure.:Wait.") == 2)
		close;
	// Through Paladin, at job 1: see the header.
	jobchange Job_Paladin;
	jobchange Job_Royal_Guard_T;
	set ADVJOB, 0;
	mes .@n$;
	mes "Welcome, Royal Guard.";
	close;
}
