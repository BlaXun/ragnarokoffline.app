//===== soul-reaper ============================================
//= Soul Reaper for pre-renewal: the Soul Linker's rebirth.
//=
//= A Soul Linker at base 99 / job 50 is reborn as a Novice.
//= At job 10 the Novice becomes a Taekwon, with the transcendent
//= class's 100 starting status points. From Taekwon job 40 the
//= reborn Taekwon becomes a Soul Reaper (job levels to 70), the
//= way a reborn Swordman becomes a Lord Knight.
//=
//= Soul Reaper is a third class in rAthena, so the change goes
//= through Soul Linker: jobchange to Soul Linker, then at once
//= to Soul Reaper. rAthena records the job level each change
//= was made at and holds the skill tree back until that many
//= points are spent in the class before. Going through Soul
//= Linker at job 1 records nothing for it, so the 69 Soul Reaper
//= points go wherever the player likes: Soul Linker skills (the
//= tree inherits them) or Soul Reaper skills.
//= Total: 9 + 49 + 69 = 127 at Taekwon job 50, a transcendent
//= class's.
//=
//= The character variable soul_reaper_reborn marks the path:
//=   1  reborn as a Novice
//=   2  a Taekwon again
//=   3  a Soul Reaper
//==============================================================

payon,154,141,5	script	Shaman Seol#sr_prere	716,{
	.@n$ = "[Shaman Seol]";
	if (Class == Job_Soul_Reaper) {
		mes .@n$;
		mes "The souls answer to you now, Soul Reaper.";
		mes "The restless dead carry gear made for your kind. Take it from them.";
		close;
	}
	if (Upper != 0) {
		mes .@n$;
		mes "This path is not one I can show you.";
		close;
	}
	switch (soul_reaper_reborn) {
	case 0:
		if (Class != Job_Soul_Linker) {
			mes .@n$;
			mes "A Soul Linker who has learned all a Soul Linker can learn may be born again, and rise as a Soul Reaper.";
			mes "Come back as a Soul Linker of base level 99 and job level 50.";
			close;
		}
		if (BaseLevel < 99 || JobLevel < 50) {
			mes .@n$;
			mes "You are not ready. Come back at base level 99 and job level 50.";
			close;
		}
		mes .@n$;
		mes "You have learned all a Soul Linker can learn.";
		mes "If you are reborn, you begin again as a Novice: level 1, no skills, no stats.";
		next;
		mes .@n$;
		mes "At job level 10 you come back to me and become a Taekwon again, stronger in body than before.";
		mes "From Taekwon job level 40 you may become a ^0000FFSoul Reaper^000000, and grow to job level 70.";
		mes "Wait until Taekwon job level 50 and you will have ten more points for Taekwon skills.";
		next;
		mes .@n$;
		mes "Come with nothing: no items, no zeny, and no skill points left to spend.";
		if (select("I am ready to be reborn.:Not yet.") == 2) {
			mes .@n$;
			mes "The souls will wait.";
			close;
		}
		if (Weight > 0 || Zeny > 0 || SkillPoint > 0) {
			mes .@n$;
			mes "You still carry something: items, zeny or unspent skill points. Come back without them.";
			close;
		}
		callfunc "F_ClearJobVar";
		soul_reaper_reborn = 1;
		jobchange Job_Novice;
		resetlvl(1);
		StatusPoint = 48;
		skill "NV_FIRSTAID",1,SKILL_PERM;
		skill "NV_TRICKDEAD",1,SKILL_PERM;
		getitem 1202,1; // Knife_
		getitem 2302,1; // Cotton_Shirt_
		mes .@n$;
		mes "Begin again. Come back at job level 10.";
		close;

	case 1:
		if (Class != Job_Novice) {
			mes .@n$;
			mes "You were reborn to walk my path, and took another. I cannot help you now.";
			close;
		}
		if (JobLevel < 10 || getskilllv("NV_BASIC") < 9) {
			mes .@n$;
			mes "Come back at job level 10, with your Basic Skill learned.";
			close;
		}
		mes .@n$;
		mes "Become a Taekwon once more.";
		if (select("I am ready.:Not yet.") == 2)
			close;
		soul_reaper_reborn = 2;
		jobchange Job_Taekwon;
		StatusPoint += 52;
		mes .@n$;
		mes "Your body remembers. Come back at Taekwon job level 40 or later.";
		close;

	case 2:
		if (Class != Job_Taekwon) {
			mes .@n$;
			mes "You were reborn to walk my path, and took another. I cannot help you now.";
			close;
		}
		if (JobLevel < 40) {
			mes .@n$;
			mes "Come back at Taekwon job level 40 or later.";
			close;
		}
		if (SkillPoint > 0) {
			mes .@n$;
			mes "Spend your skill points first.";
			close;
		}
		mes .@n$;
		mes "You may become a ^0000FFSoul Reaper^000000 now.";
		if (JobLevel < 50) {
			mes "If you wait until job level 50 you will have " + (50 - JobLevel) + " more point" + (50 - JobLevel == 1 ? "" : "s") + " for Taekwon skills.";
		}
		mes "As a Soul Reaper, your 69 job levels go to Soul Linker and Soul Reaper skills alike.";
		if (select("Make me a Soul Reaper.:Not yet.") == 2)
			close;
		soul_reaper_reborn = 3;
		// Through Soul Linker, at job 1: see the header.
		jobchange Job_Soul_Linker;
		jobchange Job_Soul_Reaper;
		mes .@n$;
		mes "Rise, Soul Reaper.";
		close;
	}
	end;
}
