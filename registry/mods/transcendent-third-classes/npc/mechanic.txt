//===== transcendent-third-classes: Mechanic ===================
//= The Blacksmith's other transcendent path.
//=
//= After the Valkyrie's rebirth, a High Merchant of job 40 or more
//= whose rebirth target is Whitesmith (ADVJOB) may become a
//= Mechanic here instead, beside the stock Whitesmith changer,
//= which is untouched. The choice is permanent.
//=
//= Mechanic is a third class in rAthena. The change goes
//= through Whitesmith and at once on to Mechanic, so
//= rAthena's third-class skill lock records Whitesmith job 1
//= and holds nothing back. The class's tree is renewal's
//= non-transcendent Mechanic tree (Novice, Merchant, Blacksmith
//= and its own skills), with the same 69 job levels a
//= Whitesmith has: a sidegrade, not an upgrade.
//==============================================================

valkyrie,42,50,5	script	Mechanic#Valkyrie	731,{
	.@n$ = "[Mechanic]";
	if (Class == Job_Mechanic_T) {
		mes .@n$;
		mes "You chose the machine. The Mado Engineer in Einbroch will put you in one once you hold its licence, and the Black Marketeer beside him sells its fuel. Axes for it are carried by the creatures of fire in the volcano, by the queen of the ant hell and by the dragon of Abyss Lake.";
		close;
	}
	if (ADVJOB != Job_Whitesmith || Class != Job_Merchant_High) {
		mes .@n$;
		mes "A Blacksmith who has been reborn chooses here: the Whitesmith's path, or mine.";
		close;
	}
	if (JobLevel < 40) {
		mes .@n$;
		mes "Come back at job level 40 or later.";
		close;
	}
	if (SkillPoint) {
		mes .@n$;
		mes "Spend your skill points first.";
		close;
	}
	mes .@n$;
	mes "A Whitesmith fights with the cart and the forge: Cart Termination, Meltdown, Maximum Power-Thrust, Upgrade Weapon. A ^0000FFMechanic^000000 leaves those behind for the Madogear: a walking machine of cannons, drills, flames and shields, fed on Magic Gear Fuel.";
	next;
	mes .@n$;
	mes "You keep every Blacksmith skill, the forge among them. You will never learn a Whitesmith skill: no Cart Termination, no Meltdown, no Maximum Power-Thrust, no Upgrade Weapon.";
	mes "Your job levels, your body and your gear are the same as a Whitesmith's.";
	next;
	mes .@n$;
	mes "^FF0000This choice is permanent.^000000";
	if (select("I take the machine.:Not yet.") == 2)
		close;
	if (select("Permanently. I am sure.:Wait.") == 2)
		close;
	// Through Whitesmith, at job 1: see the header.
	jobchange Job_Whitesmith;
	jobchange Job_Mechanic_T;
	set ADVJOB, 0;
	mes .@n$;
	mes "Welcome, Mechanic.";
	close;
}
