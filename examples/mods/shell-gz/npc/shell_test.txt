// A GM's test bench for the shell control commands. Whisper to npc:shelltest,
// with the arguments separated by # (rAthena splits a whisper to an NPC there):
//
//   near                    list the shells within 14 cells: unit id, kind, name
//   spawn#Priest{#50#Name}  make a shell beside you and take it (job, level, name)
//   grab                    take the nearest free ambient shell
//   come                    the taken shell walks to you          (unitwalk)
//   follow / stop           it follows you, or stops              (pcfollow)
//   attack                  it attacks the nearest monster        (unitattack)
//   say#text                it says something                     (unittalk)
//   free                    give it back to the engine            (population_unhold)
//   bye{#fly}               it logs out, or fly-wings away        (population_despawn)
//
// While taken, whispering the shell itself runs OnShellWhisper below, which
// answers through population_whisper. Each take lasts ten minutes.

-	script	shelltest	-1,{
OnWhisperGlobal:
	if (getgmlevel() < 60)
		end;
	.@cmd$ = @whispervar0$;
	getmapxy(.@map$, .@x, .@y);
	if (.@cmd$ == "near") {
		.@n = population_shells(.@map$, .@x, .@y, 14, .@gid, 1|2|4);
		dispbottom "shelltest: " + .@n + " shell(s) within 14 cells, vendors, held and dead included. Kind: 1 ambient, 2 vendor, 3 companion.";
		for (.@i = 0; .@i < .@n && .@i < 15; .@i++)
			dispbottom "  " + .@gid[.@i] + "  kind " + population_isshell(.@gid[.@i]) + "  " + rid2name(.@gid[.@i]);
		end;
	}
	if (.@cmd$ == "spawn") {
		.@lv = atoi(@whispervar2$);
		.@gid = population_spawn(.@map$, .@x + 2, .@y, @whispervar1$, .@lv, @whispervar3$);
		if (!.@gid) {
			dispbottom "shelltest: spawn refused (unknown job, no profile for it, name taken, or population full).";
			end;
		}
		callsub S_Take, .@gid;
		end;
	}
	if (.@cmd$ == "grab") {
		if (population_shells(.@map$, .@x, .@y, 14, .@gid) <= 0) {
			dispbottom "shelltest: no free ambient shell within 14 cells.";
			end;
		}
		callsub S_Take, .@gid[0];
		end;
	}
	if (.last == 0 || !population_isshell(.last)) {
		dispbottom "shelltest: take a shell first (spawn or grab).";
		end;
	}
	if (.@cmd$ == "come")
		unitwalk .last, .@x + 1, .@y;
	else if (.@cmd$ == "follow")
		pcfollow .last, getcharid(3);
	else if (.@cmd$ == "stop")
		pcstopfollow .last;
	else if (.@cmd$ == "attack") {
		if (getareaunits(BL_MOB, .@map$, .@x - 14, .@y - 14, .@x + 14, .@y + 14, .@mob) <= 0) {
			dispbottom "shelltest: no monster within 14 cells.";
			end;
		}
		unitattack .last, .@mob[0], 1;
	} else if (.@cmd$ == "say")
		unittalk .last, rid2name(.last) + " : " + @whispervar1$;
	else if (.@cmd$ == "free") {
		dispbottom "shelltest: unhold " + population_unhold(.last) + ".";
		.last = 0;
	} else if (.@cmd$ == "bye") {
		dispbottom "shelltest: despawn " + population_despawn(.last, @whispervar1$ == "fly") + ".";
		.last = 0;
	} else
		dispbottom "shelltest: unknown command '" + .@cmd$ + "'.";
	end;

S_Take:
	.@gid = getarg(0);
	if (!population_hold(.@gid, 600000)) {
		dispbottom "shelltest: could not take " + .@gid + ".";
		return;
	}
	population_whisperevent .@gid, strnpcinfo(3) + "::OnShellWhisper";
	.last = .@gid;
	dispbottom "shelltest: took " + rid2name(.@gid) + " (" + .@gid + ") for ten minutes.";
	return;

OnShellWhisper:
	population_whisper @shell_gid, "You said: " + @shell_msg$;
	end;
}
